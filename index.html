05
Flujo de trabajo en GitHub
Cómo usar el repo día a día

    

      

        
1. Crear el repo

        
Sube el kit al repo taller-fm con la estructura de carpetas mostrada arriba.

        
git init
git add .
git commit -m "init: estructura base TALLER FM"
git branch -M main
git remote add origin https://github.com/TU-USUARIO/taller-fm.git
git push -u origin main

      

      

        
2. Activar Projects

        
GitHub → Projects → New project → Board. Crea las 6 columnas del Kanban y vincula el repo.

        
Columnas:
· Backlog
· Planificado
· En desarrollo
· En revisión
· Listo para deploy
· Completado

      

      

        
3. Cargar la flota

        
Edita data/flota/tractos.json y remolques.json con tus unidades reales.

        
data/flota/tractos.json
data/flota/remolques.json
data/catalogo/causas.json
data/catalogo/estados.json

      

      

        
4. Registrar cambios de estado

        
Crea un Issue con el template correspondiente. El Kanban lo mueve automáticamente según las labels.

        
Labels:
· estado:inoperativo
· estado:operativo
· estado:dor
· estado:pm
· tipo:falla
· tipo:pm
· tipo:logistica
· prioridad:p1/p2/p3/p4

      

      

        
5. Desarrollar mejoras

        
Crea una rama por cada issue, desarrolla, abre un Pull Request y mergea a main.

        
git checkout -b feature/nueva-funcionalidad
# desarrolla
git add .
git commit -m "feat: nueva funcionalidad (#12)"
git push origin feature/nueva-funcionalidad
# abre PR en GitHub

      

      

        
6. Desplegar

        
El workflow de GitHub Actions publica automáticamente en GitHub Pages al mergear a main.

        
.github/workflows/deploy.yml
# se ejecuta en cada push a main
# publica en:
# https://TU-USUARIO.github.io/taller-fm/

function renderTree(){
  const paths=Object.keys(FILES).sort();

    const parts=p.split('/');

    parts.forEach((part,i)=>{
      if(i===parts.length-1){cur[part]={_path:p,_file:FILES[p]}}
      else{cur[part]=cur[part]||{};cur=cur[part]}
    });
  });
  const render=(obj,depth=0)=>{
    let html='';
    const keys=Object.keys(obj).sort((a,b)=>{
      const aDir=typeof obj[a]==='object'&&!obj[a]._path;
      const bDir=typeof obj[b]==='object'&&!obj[b]._path;
      if(aDir&&!bDir)return -1;
      if(!aDir&&bDir)return 1;
      return a.localeCompare(b);
    });
    keys.forEach(k=>{
      if(k.startsWith('_'))return;
      const val=obj[k];
      const isDir=typeof val==='object'&&!val._path;
      const indent=depth*16;
      if(isDir){
        html+=`

        const size=(f.content.length/1024).toFixed(1);
        html+=`

    });

  };

const CRCT=(()=>{const t=new Uint32Array(256);for(let n=0;n<256;n++){let c=n;for(let k=0;k<8;k++)c=c&1?0xEDB88320^(c>>>1):c>>>1;t[n]=c>>>0}return t})();

function zipBlob(list){
  const enc=new TextEncoder(),chunks=[],cent=[];let off=0;
  const d=new Date(),tm=((d.getHours()<<11)|(d.getMinutes()<<5)|(d.getSeconds()>>1))&0xFFFF,
        dd=(((d.getFullYear()-1980)<<9)|((d.getMonth()+1)<<5)|d.getDate())&0xFFFF;
  for(const f of list){
    const nm=enc.encode(f.path),dt=enc.encode(f.content),crc=crc32(dt);
    const lh=new Uint8Array(30+nm.length),lv=new DataView(lh.buffer);
    lv.setUint32(0,0x04034b50,true);lv.setUint16(4,20,true);lv.setUint16(6,0x0800,true);lv.setUint16(8,0,true);
    lv.setUint16(10,tm,true);lv.setUint16(12,dd,true);lv.setUint32(14,crc,true);lv.setUint32(18,dt.length,true);lv.setUint32(22,dt.length,true);
    lv.setUint16(26,nm.length,true);lv.setUint16(28,0,true);lh.set(nm,30);
    chunks.push(lh,dt);
    const ch=new Uint8Array(46+nm.length),cv=new DataView(ch.buffer);
    cv.setUint32(0,0x02014b50,true);cv.setUint16(4,20,true);cv.setUint16(6,20,true);cv.setUint16(8,0x0800,true);cv.setUint16(10,0,true);
    cv.setUint16(12,tm,true);cv.setUint16(14,dd,true);cv.setUint32(16,crc,true);cv.setUint32(20,dt.length,true);cv.setUint32(24,dt.length,true);
    cv.setUint16(28,nm.length,true);cv.setUint32(42,off,true);ch.set(nm,46);
    cent.push(ch);off+=lh.length+dt.length;
  }
  let cs=0;cent.forEach(c=>cs+=c.length);
  const eo=new Uint8Array(22),ev=new DataView(eo.buffer);
  ev.setUint32(0,0x06054b50,true);ev.setUint16(8,list.length,true);ev.setUint16(10,list.length,true);
  ev.setUint32(12,cs,true);ev.setUint32(16,off,true);
  return new Blob([...chunks,...cent,eo],{type:'application/zip'});
}
